cmake_minimum_required(VERSION 3.24)
project(sdi_omt VERSION 1.0.0.8 LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows MSVC x64 required")
endif()
set(OMT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/omt/Libraries/Winx64")
add_executable(sdi_omt src/main.cpp)
target_compile_features(sdi_omt PRIVATE cxx_std_20)
target_compile_definitions(sdi_omt PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(sdi_omt PRIVATE /W4 /utf-8 /EHsc)
target_include_directories(sdi_omt PRIVATE "${OMT_DIR}")
target_link_directories(sdi_omt PRIVATE "${OMT_DIR}")
target_link_libraries(sdi_omt PRIVATE libomt ole32 oleaut32)
foreach(dll libomt libvmx)
  add_custom_command(TARGET sdi_omt POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OMT_DIR}/${dll}.dll" "$<TARGET_FILE_DIR:sdi_omt>")
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(overload_queue_test tests/overload_queue.cpp)
  target_compile_features(overload_queue_test PRIVATE cxx_std_20)
  target_include_directories(overload_queue_test PRIVATE src)
  target_compile_options(overload_queue_test PRIVATE /W4 /utf-8 /EHsc)
  add_test(NAME overload_queue COMMAND overload_queue_test)
endif()
